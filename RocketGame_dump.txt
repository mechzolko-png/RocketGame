#include <random>
#include <cstdint>
#include <cmath>
#include <utility>

void StarField::generateChunk(int x, int y) {
    auto key = std::make_pair(x, y);

    if (chunks.find(key) != chunks.end()) {
        return;
    }

    std::uint32_t seed =
        static_cast<std::uint32_t>(x) * 73856093u ^
        static_cast<std::uint32_t>(y) * 19349663u ^
        0x9E3779B9u;

    std::mt19937 rng(seed);

    std::uniform_real_distribution<float> pos(0.0f, chunkSize);
    std::uniform_real_distribution<float> size(1.0f, 2.5f);
    std::uniform_int_distribution<int> brightness(150, 255);

    std::vector<Star> stars;
    stars.reserve(starsPerChunk);

    for (int i = 0; i < starsPerChunk; i++) {
        unsigned char b = static_cast<unsigned char>(brightness(rng));

        StarField::Star star = {
            {
                x * chunkSize + pos(rng),
                y * chunkSize + pos(rng)
            },
            size(rng),
            {b, b, b, 255}
        };

        stars.push_back(star);
    }

    chunks[key] = std::move(stars);
}

void StarField::update(Vector2D cameraPosition, float screenWidth, float screenHeight, float zoom) {
    if (zoom <= 0.0f) {
        return;
    }

    float halfWidth = screenWidth / (2.0f * zoom) + margin;
    float halfHeight = screenHeight / (2.0f * zoom) + margin;

    int minX = static_cast<int>(std::floor((cameraPosition.x - halfWidth) / chunkSize));
    int maxX = static_cast<int>(std::floor((cameraPosition.x + halfWidth) / chunkSize));
    int minY = static_cast<int>(std::floor((cameraPosition.y - halfHeight) / chunkSize));
    int maxY = static_cast<int>(std::floor((cameraPosition.y + halfHeight) / chunkSize));

    for (int x = minX; x <= maxX; x++) {
        for (int y = minY; y <= maxY; y++) {
            generateChunk(x, y);
        }
    }

    for (auto it = chunks.begin(); it != chunks.end();) {
        if (it->first.first < minX - 1 || it->first.first > maxX + 1 ||
            it->first.second < minY - 1 || it->first.second > maxY + 1) {
            it = chunks.erase(it);
        } else {
            ++it;
        }
    }
}

void StarField::draw() {
    for (auto& chunk : chunks) {
        for (auto& star : chunk.second) {
            DrawCircleV(
                {star.position.x, star.position.y},
                star.radius,
                star.color
            );
        }
    }
}

void GameWorld::useBullet() {
    for (int i = 0; i < bullets.size();) {
        Bullet& bullet = bullets[i];
        bool hit = false;

        for (int j = 0; j < enemies.size(); j++) {
            Enemy& enemy = enemies[j];

            if (Math::distance(enemy.position, bullet.position) <= enemy.radius + bullet.r) {
                enemy.hp -= 10;
                hit = true;
                break;
            }
        }

        if (hit) {
            bullets.erase(bullets.begin() + i);
        } else {
            i++;
        }
    }
}

        } else {
            i++;
        }
    }
}

#include "Math.h"

Vector2D Math::normalize(Vector2D vec) {
    float length = std::hypot(vec.x, vec.y);

    if (length == 0.0f || !std::isfinite(length))
        return Vector2D(0, 0);

    return Vector2D(vec.x / length, vec.y / length);
}

Vector2D Math::getVelocity(Vector2D from, Vector2D to, float speed) {
    if (!std::isfinite(speed))
        return Vector2D(0, 0);

    Vector2D direction = normalize(Vector2D(to.x - from.x, to.y - from.y));

    return Vector2D(direction.x * speed, direction.y * speed);
}

float Math::distance(Vector2D a, Vector2D b) {
    return std::hypot(b.x - a.x, b.y - a.y);
}

#include "Game/GameWorld.h"
#include "Renderer/renderer.h"
#include "Game/StarField/StarField.h"
#include <cmath>

int main() {
    InitWindow(1024,1080,"Survive To Poo To Throw To Kill ?!");
    StarField starField;
    ToggleFullscreen();
    GameWorld game = GameWorld();
    Camera2D camera = {0};
    camera.target = {10.0f, 10.0f};
    camera.offset = {
        GetScreenWidth() / 2.0f,
        GetScreenHeight() / 2.0f
    };
    camera.rotation = 0.0f;
    camera.zoom = 1.0f;

    SetTargetFPS(144);
    int FPS = 144;
    bool swap = false;

    while (!WindowShouldClose()) {
        float dt = GetFrameTime();
        Vector2 mousePos = GetScreenToWorld2D(GetMousePosition(), camera);

        RD RenderPack = game.update(dt, mousePos);

        if (IsKeyPressed(KEY_L)) {
            if (!swap) {
                SetTargetFPS(60);
                FPS = 60;
                swap = true;
            } else {
                SetTargetFPS(144);
                FPS = 144;
                swap = false;
            }
        }

        Vector2 playerPos = {
            RenderPack.player.x,
            RenderPack.player.y
        };

        float deadZone = 50.0f;

        Vector2 difference = {
            playerPos.x - camera.target.x,
            playerPos.y - camera.target.y
        };

        float distance = sqrtf(
            difference.x * difference.x +
            difference.y * difference.y
        );

        if (distance > deadZone) {
            float move = distance - deadZone;

            camera.target.x += difference.x / distance * move;
            camera.target.y += difference.y / distance * move;
        }

        starField.update(
            {camera.target.x, camera.target.y},
            GetScreenWidth(),
            GetScreenHeight(),
            camera.zoom
        );

        BeginDrawing();
        ClearBackground(BLACK);

        string TextFPS = "FPS: " + to_string(FPS);

        BeginMode2D(camera);
        DrawText("TUTORIAL:", 20, 20, 30, GREEN);
        DrawText("Controls: W A S D", 20, 60, 20, GREEN);
        DrawText(TextFPS.c_str(), 20, 80, 20, GREEN);
        DrawText("Settings: Press l to swap between 144 and 60 FPS", 20, 100, 20, GREEN);
        starField.draw();
        renderer::update(RenderPack);
        game.drawBullets();
        game.drawEnemies();
        game.player.updateSparksFromShot(dt);
        game.player.handleSparks(dt);
        EndMode2D();

        EndDrawing();
    }

    CloseWindow();

    return 0;
}
